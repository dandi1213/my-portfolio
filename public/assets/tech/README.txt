python.svg  javascript.svg  csharp.svg  php.svg  react.svg  html.svg  css.svg
rest-api.svg  mssql.svg  git.svg  selenium.svg  pandas.svg  scikit-learn.svg
esp32.svg  arduino.svg  plc.svg
